## 0. 先校准：原规划 5 处需修正的假设

| # | 原规划表述 | 实际情况 | 影响 |
|---|---|---|---|
| 1 | 「复用 `utils/throttle.py`，每用户每分钟 N 次」 | 该文件只有一个 `LoginThrottle`，是**失败计数 + 锁定**语义（`is_locked` / `record_fail` / `reset`），**没有「N 次/分钟」的放行计数 API** | 必须新增一个 `SlidingWindowRateLimiter` 类，可复用其内存字典 + `threading.Lock` + GC 的写法 |

### 2.2 限流器（`utils/throttle.py` 追加）

这是对原规划第 1 处修正的落地。**不要复用 `LoginThrottle`**：它的 `record_fail` 语义是「记一次失败」，累计 5 次后锁 15 分钟；拿它做「每分钟 5 次」会把正常用户锁死。

追加在 `login_throttle = LoginThrottle()` 之后：

```python
from collections import deque

class SlidingWindowRateLimiter:
    """滑动窗口放行计数：window 秒内同 key 最多 max_calls 次。

    与 LoginThrottle 的区别：这里是「成功也计数」的放行限流，
    不涉及锁定与失败累积，用于高频调用的外部服务入口（如语音转写）。
    """

    def __init__(self, max_calls: int = 5, window: int = 60, gc_interval: int = 300) -> None:
        self.max_calls = max_calls
        self.window = window
        self.gc_interval = gc_interval
        self._hits: dict[str, deque[float]] = {}
        self._last_gc = time.time()
        self._lock = threading.Lock()

    def _gc_if_needed(self, now: float) -> None:
        if now - self._last_gc < self.gc_interval:
            return
        cutoff = now - self.window
        self._hits = {k: v for k, v in self._hits.items() if v and v[-1] > cutoff}
        self._last_gc = now

    def allow(self, key: str) -> tuple[bool, int]:
        """返回 (是否放行, 还需等待秒数)；放行时等待秒数为 0"""
        with self._lock:
            now = time.time()
            self._gc_if_needed(now)
            dq = self._hits.setdefault(key, deque())
            cutoff = now - self.window
            while dq and dq[0] <= cutoff:
                dq.popleft()
            if len(dq) >= self.max_calls:
                return False, int(dq[0] + self.window - now) + 1
            dq.append(now)
            return True, 0


voice_rate_limiter = SlidingWindowRateLimiter(
    max_calls=VOICE_RATE_LIMIT_PER_MIN, window=60
)
```

需要在文件顶部补 `from config.settings import VOICE_RATE_LIMIT_PER_MIN`。`config/settings.py` 不反向依赖 `utils`，无循环导入风险。

### 2.3 STT Provider（新增 `ai/stt.py`）

对原规划第 3 处修正的落地：用 `segments[].avg_logprob` 换算置信度代理值。

```python
"""语音转写（STT）— Provider 抽象 + OpenAI 兼容 Whisper 实现

口径（规划 §4.2）：
- 只做「音频 → 文本」，不做任何记账语义解析；
- provider 可替换：实现 SttProvider + 在 _REGISTRY 注册一行，前端与路由零改动；
- 音频只在内存中流转，**不落磁盘、不入库**；
- 复用项目既有的 OpenAI 兼容网关模式（与 ai/llm.py 同构）。
"""
from __future__ import annotations

import logging
import math
from typing import Protocol

import httpx
from pydantic import BaseModel

from config.settings import (
    AGNES_API_KEY, OPENAI_BASE_URL, STT_API_KEY, STT_BASE_URL,
    STT_ENABLED, STT_MODEL, STT_PROVIDER, STT_TIMEOUT_SECONDS,
)

logger = logging.getLogger(__name__)


class ProviderNotConfigured(RuntimeError):
    """provider 未启用 / 配置缺失 —— 路由据此返回 ai_error"""


class SttResult(BaseModel):
    text: str
    language: str | None = None
    confidence: float | None = None


class SttProvider(Protocol):
    async def transcribe(self, audio_bytes: bytes, filename: str) -> SttResult: ...


def _confidence_from_segments(payload: dict) -> float | None:
    """Whisper 不直接给置信度，用 segments[].avg_logprob 换算代理值。

    avg_logprob 是对数概率（通常 -1.0 ~ 0），exp() 后映射到 0~1；
    同时用 no_speech_prob 压低「整段是静音/噪声」的可信度。
    """
    segs = payload.get("segments") or []
    if not segs:
        return None
    probs = [math.exp(s.get("avg_logprob", 0.0)) for s in segs]
    base = sum(probs) / len(probs)
    no_speech = max((s.get("no_speech_prob", 0.0) for s in segs), default=0.0)
    return round(max(0.0, min(1.0, base * (1.0 - no_speech))), 4)


class OpenAIWhisperProvider:
    """POST {base_url}/audio/transcriptions（OpenAI 兼容 multipart）"""

    def __init__(self, base_url: str, api_key: str, model: str, timeout: float = 30.0):
        self._url = base_url.rstrip("/") + "/audio/transcriptions"
        self._api_key = api_key
        self._model = model
        self._timeout = timeout

    async def transcribe(self, audio_bytes: bytes, filename: str) -> SttResult:
        files = {"file": (filename, audio_bytes, "application/octet-stream")}
        data = {"model": self._model, "response_format": "verbose_json"}
        headers = {"Authorization": f"Bearer {self._api_key}"}
        async with httpx.AsyncClient(timeout=self._timeout) as client:
            resp = await client.post(self._url, headers=headers, files=files, data=data)
            resp.raise_for_status()
            payload = resp.json()
        return SttResult(
            text=(payload.get("text") or "").strip(),
            language=payload.get("language"),
            confidence=_confidence_from_segments(payload),
        )


_REGISTRY: dict[str, object] = {}


def register(name: str):
    def deco(factory):
        _REGISTRY[name] = factory
        return factory
    return deco


@register("openai_whisper")
def _build_openai_whisper() -> SttProvider:
    if not STT_API_KEY:
        raise ProviderNotConfigured("STT_API_KEY 未配置")
    return OpenAIWhisperProvider(
        base_url=STT_BASE_URL or OPENAI_BASE_URL,
        api_key=STT_API_KEY or AGNES_API_KEY,
        model=STT_MODEL,
        timeout=STT_TIMEOUT_SECONDS,
    )


_provider: SttProvider | None = None


def get_stt_provider() -> SttProvider:
    """按 STT_PROVIDER 取 provider；未启用/未注册 → ProviderNotConfigured"""
    global _provider
    if _provider is None:
        if not STT_ENABLED:
            raise ProviderNotConfigured("语音识别未启用")
        factory = _REGISTRY.get(STT_PROVIDER)
        if factory is None:
            raise ProviderNotConfigured(f"未知的 STT provider: {STT_PROVIDER}")
        _provider = factory()          # type: ignore[operator]
    return _provider
```

**关于临时文件：建议不要写。** 原规划要求「写 `tempfile.mkdtemp` → `try/finally` 删除」。既然 `transcribe()` 收的是 `bytes`，直接把 `await file.read()` 的内存字节交给 httpx 即可——**少一个失败面**（不用怕 finally 漏删、不用怕并发文件名冲突、不用怕容器只读文件系统）。若未来接入必须读文件路径的 provider（如本地 funasr CLI），由该 provider 自己落临时文件，抽象层保持不变。

### 2.4 端点（`routers/ai.py` 追加）

放在 `# ── 反馈 / 统计 ──` 之前，作为新的 `# ── 语音 ──` 段。顶部 import 需补 `File, UploadFile`。

```python
# ── 语音 ────────────────────────────────────────────────────────────────

_ALLOWED_AUDIO_EXT = {"mp3", "m4a", "wav", "aac", "webm", "ogg"}


@router.get("/voice/config", summary="语音功能开关")
async def voice_config(user_id: int = Depends(get_current_user)):
    """6.18 前端据此决定是否渲染话筒按钮（原规划遗漏的开关下发）"""
    return {"enabled": STT_ENABLED, "maxSeconds": STT_MAX_SECONDS}


@router.post("/voice/transcribe", summary="语音转写")
async def voice_transcribe(
    file: UploadFile = File(...),
    user_id: int = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """6.19 语音转写：裸 {text, language, confidence}；失败 ai_error

    音频只在内存流转，不落盘、不入库。
    """
    if not STT_ENABLED:
        return ai_error("语音识别未启用")

    # 1) 节流（按用户，滑动窗口）
    ok, retry_after = voice_rate_limiter.allow(f"voice:{user_id}")
    if not ok:
        return ai_error(f"说话太频繁，请 {retry_after} 秒后再试")

    # 2) 扩展名白名单 + 大小校验
    filename = (file.filename or "audio.m4a").strip()
    ext = filename.rsplit(".", 1)[-1].lower() if "." in filename else ""
    if ext not in _ALLOWED_AUDIO_EXT:
        return ai_error("不支持的音频格式，请使用 mp3 / m4a / wav / aac")

    data = await file.read()
    if not data:
        return ai_error("音频内容为空")
    if len(data) > STT_MAX_SIZE_MB * 1024 * 1024:
        return ai_error(f"音频过大，请控制在 {STT_MAX_SIZE_MB}MB 以内")

    # 3) 转写（异常不回显底层报错）
    try:
        provider = get_stt_provider()
        result = await provider.transcribe(data, filename)
    except ProviderNotConfigured:
        return ai_error("语音识别未启用")
    except Exception:
        logger.exception("voice transcribe failed user_id=%s size=%s", user_id, len(data))
        return ai_error("语音识别暂不可用，请稍后重试")

    return {"text": result.text, "language": result.language, "confidence": result.confidence}
```

要点：
- 遵守 AI 模块铁律——裸返回、业务失败 `ai_error`、不抛 `HTTPException`；
- `STT_MAX_SECONDS` 是**客户端约束**（`recorder.start({duration})`），服务端拿不到音频时长，不做校验（Whisper 侧兜底）；
- 转写空文本返回 `{"text": ""}`，由前端提示「没听清」，**不在这里返 error**（空文本不是故障）。

### 4.2 上传封装（新增 `api/voice.js`）

**不能复用 `api/request.js`**——`uni.uploadFile` 的 `res.data` 是字符串，且不走 `request` 的拦截器。但要复用它的 401 处理口径：

```js
import BASE_URL from './config'

/**
 * 语音转写。uni.uploadFile 的响应体是字符串且不走 request.js 拦截器，
 * 因此单独封装，仅对齐其 401 / 错误提示口径。
 * @returns {Promise<{text:string, language?:string, confidence?:number}>}
 */
export const transcribeVoice = (filePath) => {
	return new Promise((resolve, reject) => {
		const token = uni.getStorageSync('token')
		uni.uploadFile({
			url: BASE_URL + '/ai/voice/transcribe',
			filePath,
			name: 'file',
			header: { 'Authorization': token ? `Bearer ${token}` : '' },
			timeout: 30000,
			success: (res) => {
				if (res.statusCode === 401) {
					uni.removeStorageSync('token')
					uni.reLaunch({ url: '/pages/login/index' })
					reject(new Error('未登录或登录已过期'))
					return
				}
				let body = null
				try { body = JSON.parse(res.data) } catch (e) { body = null }
				if (!body) { reject(new Error('语音识别返回异常')); return }
				if (body.error) { reject(new Error(body.error)); return }
				resolve(body)
			},
			fail: (err) => {
				const msg = (err && err.errMsg) || ''
				reject(new Error(msg.includes('timeout') ? '识别超时，请重试' : '语音上传失败'))
			}
		})
	})
}

/** 语音开关（前端据此隐藏话筒） */
export const getVoiceConfig = () => request({ url: '/ai/voice/config', method: 'GET' })
```

### 4.4 录音（`pages/ai-chat/index.vue`）

**关键陷阱：`uni.getRecorderManager()` 是全局单例，回调必须只注册一次。** 若写在 `startRecord()` 里，每次录音都会叠加一个 `onStop`，第二次录音会触发两次转写。

```js
const recorder = uni.getRecorderManager()
const recording = ref(false)
const transcribing = ref(false)
const recordSeconds = ref(0)
let recordTimer = null
let recordStartAt = 0

// 模块加载时注册一次（不要放进 startRecord）
recorder.onStart(() => {
	recording.value = true
	recordSeconds.value = 0
	recordStartAt = Date.now()
	recordTimer = setInterval(() => {
		recordSeconds.value = Math.floor((Date.now() - recordStartAt) / 1000)
		if (recordSeconds.value >= 60) stopRecord()      // 兜底自动停
	}, 1000)
})

recorder.onStop((res) => {
	clearInterval(recordTimer)
	recording.value = false
	if (!res || !res.tempFilePath) return
	if (Date.now() - recordStartAt < 500) {             // 误触：按一下就松
		uni.showToast({ title: '说话时间太短', icon: 'none' })
		return
	}
	handleTranscribe(res.tempFilePath)
})

recorder.onError(() => {
	clearInterval(recordTimer)
	recording.value = false
	uni.showToast({ title: '无法录音，请检查麦克风权限', icon: 'none' })
})

function startRecord() {
	if (loading.value || streaming.value || transcribing.value || !isLoggedIn.value) return
	recorder.start({
		duration: 60000, sampleRate: 16000, numberOfChannels: 1,
		// H5 端 MediaRecorder 只认 webm/opus；小程序/App 用 mp3
		// #ifdef H5
		format: 'webm',
		// #endif
		// #ifndef H5
		format: 'mp3',
		// #endif
		bitRate: 48000
	})
}

function stopRecord() {
	try { recorder.stop() } catch (e) {}
}

async function handleTranscribe(tempFilePath) {
	transcribing.value = true
	uni.showLoading({ title: '识别中...', mask: true })
	try {
		const res = await transcribeVoice(tempFilePath)
		const text = (res.text || '').trim()
		if (!text) {
			uni.showToast({ title: '没听清，请再说一次', icon: 'none' })
			return
		}
		// 回填输入框，用户可改后再发送（规划 §4.1 的核心交互）
		inputText.value = inputText.value ? inputText.value + text : text
		if (res.confidence != null && res.confidence < 0.4) {
			uni.showToast({ title: '识别可能不准确，请核对', icon: 'none' })
		}
	} catch (e) {
		uni.showModal({
			title: '语音识别失败',
			content: e.message || '请稍后重试',
			confirmText: '改为键盘输入',
			showCancel: false
		})
	} finally {
		uni.hideLoading()
		transcribing.value = false
	}
}
```

### 4.5 确认卡（新增 `components/VoiceConfirmPanel.vue`）

数据来源是 `dryRun=true` 的返回：`preview.items[]`，每项含 `index / amount / amountRaw / categoryName / categoryId / date / dateExpr / remark / confidence`。

提交时只回传**改过的**字段，并用服务端给的 `index` 定位：

```js
// 卡片内部：行内编辑后收集 diff
const buildOverrides = (rows) =>
	rows
		.map((r) => {
			const ov = { index: r.index }        // ← 用服务端的 index，不用行号
			if (r.edited.amount !== r.original.amount) ov.amount = Number(r.edited.amount)
			if (r.edited.categoryId !== r.original.categoryId) ov.categoryId = r.edited.categoryId
			if (r.edited.date !== r.original.date) ov.date = r.edited.date
			if (r.edited.remark !== r.original.remark) ov.remark = r.edited.remark
			return Object.keys(ov).length > 1 ? ov : null
		})
		.filter(Boolean)

// 确认入账：同一个 clientMsgId，dryRun=false
await aiBookkeeping({
	sessionId: sessionId.value,
	message: rawText,
	clientMsgId: pendingClientMsgId,     // ← 预览时生成，确认时复用
	source: 'voice',
	overrides: buildOverrides(rows)

```

**`clientMsgId` 必须跨「预览 → 确认」两次请求保持一致**，否则双击确认会入账两笔。由于 `dry_run=true` 不写日志、不消耗幂等键，首次真正落库发生在确认那次，重复点击会被 `find_log_by_client_msg` 拦成 duplicate——这正是期望行为。

## 6. 风险补充

- **两次 LLM 调用**（预览 + 确认）：原规划已识别。除「按 `requestId` 缓存 5 分钟」外，还有一个更省的做法——预览时把 `parsed` 结果连同 `dedup_hash` 暂存到 `ai_bookkeeping_logs`（`status=0` 草稿态，`action='preview'`），确认时按 `requestId` 读回直接落库，**彻底省掉第二次 LLM 调用**。代价是确认逻辑要新增一条「从草稿直接入账」的路径，复杂度高于缓存方案，MVP 阶段不建议。
- **Whisper 中文口语数字识别**：`normalizers._cn_to_int` 只支持到十位（注释明确「不支持百以上，交给 LLM」）。语音输入更容易出现「一千二」这类说法，若识别为「1200」则正常，若识别为「一千二」则 `_RE_CN` 匹配不到 → 走 `clarify` 追问，**不会错记**，符合「绝不猜测」的口径。可作为已知限制写入论文。
- **限流器单进程**：`voice_rate_limiter` 与 `login_throttle` 一样是进程内字典，多 worker 部署时各进程独立计数。仓库当前单 worker 运行，生产应换 Redis。
- **`STT_BASE_URL` 默认继承 `OPENAI_BASE_URL`**：agnes 网关是否支持 `/audio/transcriptions` 需**先实测一次 curl**。若不支持，改 `STT_BASE_URL` 指向 OpenAI 官方或自建 faster-whisper 服务即可，代码零改动——这正是 Provider 抽象的价值。
